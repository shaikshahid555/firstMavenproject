pipeline {
    agent any

    stages {

        stage('Git Checkout') {
            steps {
                git branch: 'master',
                    url: 'https://github.com/shaikshahid555/firstMavenproject.git'
            }
        }

        stage('Build') {
            steps {
                bat 'mvn clean package'
            }
        }

        stage('Run App') {
            steps {
                bat 'java -cp target/classes com.example.App'
            }
        }
    }
}
