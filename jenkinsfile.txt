pipeline {
    agent any

    stages {

        stage('Git Checkout') {
            steps {
                git branch: 'master',
                    url: 'https://github.com/shaikshahid555/firstMavenproject.git'
            }
        }

        stage('Build') {
            steps {
                bat 'mvn clean package'
            }
        }

        stage('Run Test Cases') {
            steps {
                bat 'mvn test'
            }
        }

        stage('Completed') {
            steps {
                echo 'Test cases executed successfully'
            }
        }
    }
}
